# Dedicated male-mother series. These fixtures test immediate runtime roles,
# persistence, reconstruction, and male-carrier pregnancy support.

agp_native_mother_test.0001 = {
	type = character_event
	orphan = yes
	title = agp_native_mother_test.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = agp_native_mother_test_set_mother_ok }
				desc = agp_native_mother_test.0001.desc.success
			}
			desc = agp_native_mother_test.0001.desc.failure
		}
	}
	theme = family
	left_portrait = scope:mother_test_father
	center_portrait = scope:mother_test_child
	right_portrait = scope:mother_test_mother
	immediate = {
		remove_character_flag = agp_native_mother_test_set_mother_ok
		create_character = { gender = male age = 34 employer = root save_scope_as = mother_test_father }
		create_character = { gender = male age = 32 employer = root save_scope_as = mother_test_mother }
		create_character = { gender = male age = 12 employer = root save_scope_as = mother_test_child }
		scope:mother_test_child = {
			set_father = scope:mother_test_father
			set_mother = scope:mother_test_mother
		}
		if = {
			limit = {
				scope:mother_test_child.father = { this = scope:mother_test_father }
				scope:mother_test_child.mother = { this = scope:mother_test_mother }
			}
			add_character_flag = agp_native_mother_test_set_mother_ok
		}
	}
	option = { name = agp_native_mother_test.close }
}
