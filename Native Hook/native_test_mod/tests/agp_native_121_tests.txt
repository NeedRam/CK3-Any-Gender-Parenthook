# Run only in disposable games: scripted_tests agp_native_121
# This uses the 1.21 tests/_tests.info schema. Each expectation is evaluated
# against native character scopes, rather than the event's displayed text.
agp_native_121_setters_and_reassignment = {
    character_target = 122
    assert = { exists = root }
    effect = {
        create_character = { gender = female age = 40 employer = root save_scope_as = old_father }
        create_character = { gender = female age = 42 employer = root save_scope_as = new_father }
        create_character = { gender = female age = 45 employer = root save_scope_as = biological_father }
        create_character = { gender = male age = 40 employer = root save_scope_as = old_mother }
        create_character = { gender = male age = 42 employer = root save_scope_as = new_mother }
        create_character = { gender = male age = 45 employer = root save_scope_as = biological_mother }
        create_character = { gender = female age = 1 employer = root save_scope_as = test_child }
        scope:test_child = {
            set_father = scope:old_father
            set_mother = scope:old_mother
            set_real_father = scope:biological_father
            set_real_mother = scope:biological_mother
            set_father = scope:new_father
            set_mother = scope:new_mother
            add_character_flag = agp_native_121_reassignment_child
        }
        scope:new_father = { add_character_flag = agp_native_121_expected_father }
        scope:new_mother = { add_character_flag = agp_native_121_expected_mother }
        scope:biological_father = { add_character_flag = agp_native_121_expected_real_father }
        scope:biological_mother = { add_character_flag = agp_native_121_expected_real_mother }
    }
    assert = {
        scope:test_child.father = { this = scope:new_father }
        scope:test_child.mother = { this = scope:new_mother }
        scope:test_child.real_father = { this = scope:biological_father }
        scope:test_child.real_mother = { this = scope:biological_mother }
    }
    effect = { debug_log = "AGP_ACCEPTANCE_PASS agp_native_121_setters_and_reassignment" }
}

agp_native_121_create_and_family = {
    character_target = 122
    assert = { exists = root }
    effect = {
        create_character = { gender = female age = 65 employer = root save_scope_as = grandparent }
        create_character = { gender = female age = 40 employer = root father = scope:grandparent save_scope_as = test_father }
        create_character = { gender = male age = 38 employer = root save_scope_as = test_mother }
        create_character = { gender = female age = 1 employer = root father = scope:test_father mother = scope:test_mother save_scope_as = test_child }
        create_character = { gender = female age = 30 employer = root save_scope_as = unrelated }
    }
    assert = {
        scope:test_child.father = { this = scope:test_father }
        scope:test_child.mother = { this = scope:test_mother }
        scope:test_child.father.father = { this = scope:grandparent }
        scope:test_child = { is_close_family_of = scope:test_father is_close_family_of = scope:test_mother is_close_family_of = scope:grandparent }
        scope:grandparent = { is_close_family_of = scope:test_child }
        scope:unrelated = { NOT = { is_close_family_of = scope:test_child } }
    }
    effect = { debug_log = "AGP_ACCEPTANCE_PASS agp_native_121_create_and_family" }
}

agp_reload_121_ordinary_controls = {
    character_target = 122
    effect = {
        create_character = { gender = male age = 40 employer = root save_scope_as = ordinary_father }
        create_character = { gender = female age = 40 employer = root save_scope_as = ordinary_mother }
        create_character = { gender = male age = 45 employer = root save_scope_as = ordinary_real_father }
        create_character = { gender = female age = 45 employer = root save_scope_as = ordinary_real_mother }
        create_character = { gender = female age = 65 employer = root save_scope_as = paternal_father }
        create_character = { gender = male age = 65 employer = root save_scope_as = paternal_mother }
        create_character = { gender = female age = 65 employer = root save_scope_as = maternal_father }
        create_character = { gender = male age = 65 employer = root save_scope_as = maternal_mother }
        scope:ordinary_father = { set_father = scope:paternal_father set_mother = scope:paternal_mother }
        scope:ordinary_mother = { set_father = scope:maternal_father set_mother = scope:maternal_mother }
        create_character = {
            gender = female age = 1 employer = root
            father = scope:ordinary_father mother = scope:ordinary_mother
            save_scope_as = ordinary_child
        }
        scope:ordinary_child = {
            set_real_father = scope:ordinary_real_father
            set_real_mother = scope:ordinary_real_mother
        }
    }
    assert = {
        scope:ordinary_child = {
            father = scope:ordinary_father
            mother = scope:ordinary_mother
            real_father = scope:ordinary_real_father
            real_mother = scope:ordinary_real_mother
            is_close_family_of = scope:ordinary_father
            is_close_family_of = scope:ordinary_mother
            father = { father = scope:paternal_father mother = scope:paternal_mother }
            mother = { father = scope:maternal_father mother = scope:maternal_mother }
            is_close_family_of = scope:paternal_father
            is_close_family_of = scope:paternal_mother
            is_close_family_of = scope:maternal_father
            is_close_family_of = scope:maternal_mother
        }
        scope:paternal_father = { is_close_family_of = scope:ordinary_child }
        scope:paternal_mother = { is_close_family_of = scope:ordinary_child }
        scope:maternal_father = { is_close_family_of = scope:ordinary_child }
        scope:maternal_mother = { is_close_family_of = scope:ordinary_child }
    }
    effect = { debug_log = "AGP_ACCEPTANCE_PASS agp_reload_121_ordinary_controls" }
}
